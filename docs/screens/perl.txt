# words.pl -- the ten words a text uses most, with how often.
#
#     perl words.pl <file> ...
#
# Reads the files named, or standard input, and counts every word in a
# hash, ignoring case.

while (<>) {
    foreach $word (split(/\W+/, $_)) {
        $count{"\L$word"}++ if $word =~ /^[a-zA-Z]/;
    }
}
@words = sort { $count{$b} <=> $count{$a} || $a cmp $b } keys(%count);
foreach $word (@words[0..9]) {
    printf("%6d  %s\n", $count{$word}, $word);
}

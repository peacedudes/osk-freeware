options intsize=4
      program factorial
c
c *** n! = 1 * 2 * 3 * ... * n, the factorial of n.
c *** Twelve is as far as it goes in a 32-bit integer: 13! overflows.
c
      implicit none
      integer   n, i
      integer*4 f

      type *,'   n         n!'
      do n = 1, 12
        f = 1
        do i = 2, n
          f = f * i
        end do
        type *, n, f
      end do
      end
